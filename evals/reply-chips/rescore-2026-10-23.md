## Reproduction

From the repository root with Bun 1.4.3, this reads only the saved report and imports the production helpers:

```sh
bun - <<'TS'
import { readFileSync } from "node:fs";
import { hasProseAnswer } from "./openbot/scripts/evals/reply-answer";
import { hasShortReplyChips } from "./openbot/scripts/evals/reply-chips";

const source = readFileSync("evals/reply-chips/results-2026-10-23.md", "utf8");
const rows = source.split("\n")
  .filter((line) => /^\| (?:Should|Is|Would|For) /.test(line))
  .map((line, index) => {
    const cells = line.split("|").slice(1, -1).map((cell) => cell.trim());
    if (cells.length !== 4) throw new Error(`Unexpected row ${index + 1}`);
    const [ask, chipText, , answer] = cells;
    if (!ask || !answer || chipText === undefined) throw new Error("Missing fields");
    const chips = chipText ? chipText.split(" / ") : [];
    return {
      row: index + 1,
      shortChips: hasShortReplyChips(chips),
      collapsed: !hasProseAnswer(answer.split(" / "), chips),
    };
  });
if (rows.length !== 10) throw new Error(`Expected 10 rows, got ${rows.length}`);
console.log(JSON.stringify({
  qualified: rows.filter((row) => row.shortChips).length,
  collapsed: rows.filter((row) => row.collapsed).length,
  rows,
}, null, 2));
TS
```
